РћС‚РґРµР»СЊРЅС‹Р№ РјРѕРґСѓР»СЊ, РЅРµ Р·Р°РјРµРЅСЏСЋС‰РёР№ moving_average.py:
- СЃРѕС…СЂР°РЅСЏРµС‚ РїСѓР±Р»РёС‡РЅС‹Р№ РєРѕРЅС‚СЂР°РєС‚ (Sequence[float] -> list[float]);
- СЂРµР°Р»РёР·СѓРµС‚ SMA, CMA, EMA Рё WMA С‡РµСЂРµР· deque(maxlen=...);
- СѓРґРѕР±РµРЅ РґР»СЏ РїРѕС‚РѕРєРѕРІРѕР№ РѕР±СЂР°Р±РѕС‚РєРё Рё РґР»СЏ СЃСЂР°РІРЅРµРЅРёСЏ СЃ В«РёРЅРґРµРєСЃРЅРѕР№В» РІРµСЂСЃРёРµР№.
"""
from __future__ import annotations

from collections import deque
from typing import Deque, List, Sequence


def _ensure_not_empty(data: Sequence[float]) -> None:
    if not data:
        raise ValueError("Р СЏРґ РїСѓСЃС‚: СЂР°СЃС‡С‘С‚ СЃРєРѕР»СЊР·СЏС‰РµР№ СЃСЂРµРґРЅРµР№ РЅРµРІРѕР·РјРѕР¶РµРЅ")


def _check_window(window: int, n: int) -> None:
    if not isinstance(window, int) or isinstance(window, bool):
        raise ValueError(f"window РґРѕР»Р¶РµРЅ Р±С‹С‚СЊ С†РµР»С‹Рј, РїРѕР»СѓС‡РµРЅРѕ: {window!r}")
    if window < 1:
        raise ValueError(f"window РґРѕР»Р¶РµРЅ Р±С‹С‚СЊ >= 1, РїРѕР»СѓС‡РµРЅРѕ: {window}")
    if window > n:
        raise ValueError(
            f"window ({window}) Р±РѕР»СЊС€Рµ РґР»РёРЅС‹ СЂСЏРґР° ({n}) вЂ” SMA РЅРµРІРѕР·РјРѕР¶РЅР°"
        )


def _check_alpha(alpha: float) -> None:
    if not (0 < alpha <= 1):
        raise ValueError(f"alpha РґРѕР»Р¶РµРЅ Р±С‹С‚СЊ РІ (0, 1], РїРѕР»СѓС‡РµРЅРѕ: {alpha}")


def simple_moving_average_deque(
    data: Sequence[float],
    window: int,
) -> List[float]:
    """SMA С‡РµСЂРµР· deque(maxlen=window)."""
    _ensure_not_empty(data)
    _check_window(window, len(data))

    buf: Deque[float] = deque(maxlen=window)
    running = 0.0
    result: List[float] = []

    for i, x in enumerate(data):
        if len(buf) == window:
            running -= buf[0]
        buf.append(x)
        running += x
        if i >= window - 1:
            result.append(running / window)
    return result


def centered_moving_average_deque(
    data: Sequence[float],
    window: int,
) -> List[float]:
    """Р¦РµРЅС‚СЂРёСЂРѕРІР°РЅРЅРѕРµ MA: РѕРєРЅРѕ РЅРµС‡С‘С‚РЅРѕРµ, СЂРµР·СѓР»СЊС‚Р°С‚ РІС‹СЂРѕРІРЅРµРЅ РїРѕ С†РµРЅС‚СЂСѓ."""
    _ensure_not_empty(data)
    _check_window(window, len(data))
    if window % 2 == 0:
        raise ValueError(
            f"Р¦РµРЅС‚СЂРёСЂРѕРІР°РЅРЅРѕРµ MA С‚СЂРµР±СѓРµС‚ РЅРµС‡С‘С‚РЅРѕРіРѕ window, РїРѕР»СѓС‡РµРЅРѕ: {window}"
        )

    buf: Deque[float] = deque(maxlen=window)
    running = 0.0
    result: List[float] = []

    for x in data:
        if len(buf) == window:
            running -= buf[0]
        buf.append(x)
        running += x
        if len(buf) == window:
            result.append(running / window)

    return result


def exponential_moving_average_deque(
    data: Sequence[float],
    alpha: float,
) -> List[float]:
    """EMA С‡РµСЂРµР· ring-buffer deque(maxlen=1)."""
    _ensure_not_empty(data)
    _check_alpha(alpha)

    buf: Deque[float] = deque(maxlen=1)
    prev: float = float(data[0])
    result: List[float] = [prev]
    buf.append(prev)

    for x in data[1:]:
        prev = alpha * x + (1.0 - alpha) * prev
        result.append(prev)
        buf.append(prev)
    return result


def weighted_moving_average_deque(
    data: Sequence[float],
    window: int,
) -> List[float]:
    """WMA СЃ Р»РёРЅРµР№РЅС‹РјРё РІРµСЃР°РјРё 1, 2, ..., window РЅР° РѕСЃРЅРѕРІРµ deque."""
    _ensure_not_empty(data)
    _check_window(window, len(data))

    buf: Deque[float] = deque(maxlen=window)
    weights = list(range(1, window + 1))
    weight_sum = window * (window + 1) / 2.0
    result: List[float] = []

    for x in data:
        buf.append(x)
        if len(buf) == window:
            weighted_val = sum(val * w for val, w in zip(buf, weights)) / weight_sum
            result.append(weighted_val)
    return result

import time
from typing import Dict, List, Sequence

from .deque_average import simple_moving_average_deque


def check_window(window: int, n: int) -> None:
    """Р’Р°Р»РёРґР°С†РёСЏ СЂР°Р·РјРµСЂР° РѕРєРЅР°."""
    if not isinstance(window, int) or isinstance(window, bool):
        raise ValueError(f"window РґРѕР»Р¶РµРЅ Р±С‹С‚СЊ С†РµР»С‹Рј С‡РёСЃР»РѕРј, РїРѕР»СѓС‡РµРЅРѕ: {window!r}")
    if window < 1:
        raise ValueError(f"window РґРѕР»Р¶РµРЅ Р±С‹С‚СЊ >= 1, РїРѕР»СѓС‡РµРЅРѕ: {window}")
    if window > n:
        raise ValueError(f"window ({window}) Р±РѕР»СЊС€Рµ РґР»РёРЅС‹ СЂСЏРґР° ({n})")


def simple_moving_average(data: Sequence[float], window: int) -> List[float]:
    """SMA С‡РµСЂРµР· Р±РµРіСѓС‰СѓСЋ СЃСѓРјРјСѓ РїРѕ СЃРїРёСЃРєСѓ (O(n))."""
    if not data:
        raise ValueError("Р СЏРґ РїСѓСЃС‚")
    check_window(window, len(data))

    running = sum(data[:window])
    result = [running / window]
    for i in range(window, len(data)):
        running += data[i] - data[i - window]
        result.append(running / window)
    return result


def weighted_moving_average(data: Sequence[float], window: int) -> List[float]:
    """Р’Р·РІРµС€РµРЅРЅР°СЏ СЃРєРѕР»СЊР·СЏС‰Р°СЏ СЃСЂРµРґРЅСЏСЏ (WMA) СЃ Р»РёРЅРµР№РЅС‹РјРё РІРµСЃР°РјРё 1, 2, ..., window."""
    if not data:
        raise ValueError("Р СЏРґ РїСѓСЃС‚")
    check_window(window, len(data))

    weights = list(range(1, window + 1))
    weight_sum = window * (window + 1) / 2.0
    result: List[float] = []

    for i in range(len(data) - window + 1):
        window_slice = data[i : i + window]
        weighted_val = sum(x * w for x, w in zip(window_slice, weights)) / weight_sum
        result.append(weighted_val)

    return result


def benchmark_moving_average(
    data: Sequence[float], window: int, repeats: int = 1000
) -> Dict[str, float]:
    """Р—Р°РјРµСЂСЏРµС‚ СЃСЂРµРґРЅРµРµ РІСЂРµРјСЏ РІС‹РїРѕР»РЅРµРЅРёСЏ SMA РґР»СЏ СЂРµР°Р»РёР·Р°С†РёР№ РЅР° list Рё deque."""
    check_window(window, len(data))

    t0 = time.perf_counter()
    for _ in range(repeats):
        simple_moving_average(data, window)
    t_list = (time.perf_counter() - t0) / repeats

    t0 = time.perf_counter()
    for _ in range(repeats):
        simple_moving_average_deque(data, window)
    t_deque = (time.perf_counter() - t0) / repeats

    return {"list": t_list, "deque": t_deque}

        raise ValueError("Р СЏРґ РїСѓСЃС‚: СЂР°СЃС‡С‘С‚ СЃС‚Р°С‚РёСЃС‚РёРєРё РЅРµРІРѕР·РјРѕР¶РµРЅ")


def mean(data: Sequence[float]) -> float:
    """РЎСЂРµРґРЅРµРµ Р°СЂРёС„РјРµС‚РёС‡РµСЃРєРѕРµ."""
    _ensure_not_empty(data)
    return sum(data) / len(data)


def median(data: Sequence[float]) -> float:
    """РњРµРґРёР°РЅР°."""
    _ensure_not_empty(data)
    s = sorted(data)
    k = len(s) // 2
    if len(s) % 2:
        return float(s[k])
    return (s[k - 1] + s[k]) / 2.0


def spread(data: Sequence[float]) -> float:
    """Р Р°Р·РјР°С… СЂСЏРґР°: max - min."""
    _ensure_not_empty(data)
    return float(max(data) - min(data))


def stdev(data: Sequence[float]) -> float:
    """РЎС‚Р°РЅРґР°СЂС‚РЅРѕРµ РѕС‚РєР»РѕРЅРµРЅРёРµ (РЎРљРћ). Р”Р»СЏ РѕРґРёРЅР°РєРѕРІС‹С… Р·РЅР°С‡РµРЅРёР№ РІРѕР·РІСЂР°С‰Р°РµС‚ 0.0."""
    _ensure_not_empty(data)
    if len(data) < 2:
        return 0.0
    m = mean(data)
    variance = sum((x - m) ** 2 for x in data) / (len(data) - 1)
    return math.sqrt(variance)


def mode(data: Sequence[float]) -> List[float]:
    """РњРѕРґР° СЂСЏРґР° С‡РµСЂРµР· Counter. Р’РѕР·РІСЂР°С‰Р°РµС‚ СЃРїРёСЃРѕРє РІСЃРµС… РјРѕРґ."""
    _ensure_not_empty(data)
    counts = Counter(data)
    max_freq = max(counts.values())
    return sorted(float(val) for val, freq in counts.items() if freq == max_freq)


def summarize(data: Sequence[float]) -> Dict[str, Any]:
    """РЎРІРѕРґРЅС‹Р№ РѕС‚С‡С‘С‚ РїРѕ СЃС‚Р°С‚РёСЃС‚РёРєР°Рј Р’Р°СЂРёР°РЅС‚Р° 2."""
    _ensure_not_empty(data)
    return {
        "count": len(data),
        "mean": mean(data),
        "median": median(data),
        "spread": spread(data),
        "stdev": stdev(data),
        "mode": mode(data),
        "min": float(min(data)),
        "max": float(max(data)),
    }

def build_series(kind: str, n: int, seed: int) -> List[float]:
    """Р“РµРЅРµСЂРёСЂСѓРµС‚ СЂСЏРґ Р·Р°РґР°РЅРЅРѕРіРѕ С‚РёРїР°."""

    if isinstance(a, list):
        return len(a) == len(b) and all(_approx(x, y, eps) for x, y in zip(a, b))
